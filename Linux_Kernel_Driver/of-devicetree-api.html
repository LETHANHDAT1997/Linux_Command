device tree api reference

Đọc dữ liệu từ Device Tree trong driver kernel

Họ hàm of_* mà driver dùng để lấy thông tin từ node device tree, gọi bên trong probe() — khác với of_platform_*, chuyện đó nằm ở cuối trang.

mpu6050@68 {
    compatible = "ledat,mpu6050";
    reg = <0x68>;
    invensense,gyro-range = <500>;
};
Node này chính là thứ nằm sau np trong mọi ví dụ bên dưới — client->dev.of_node trỏ thẳng vào đây.

bảng so sánh DT khai gì → driver lấy bằng hàm nào

of_node bắt đầu từ đâu

Với driver I2C client (như MPU6050), device_node không cần tự đi tìm — bus core đã gán sẵn vào client->dev.of_node lúc match compatible thành công.

#include <linux/of.h>

static int mpu6050_probe(struct i2c_client *client)
{
    struct device_node *np = client->dev.of_node;

    if (!np)
        return -ENODEV; /* driver này chỉ hỗ trợ instantiate qua DT */
    ...
}

property — số & mảng

Họ hàm phổ biến nhất. Đọc một số nguyên (hoặc mảng số) khai trong DT, có fallback về mặc định nếu property không tồn tại.

int of_property_read_u32(const struct device_node *np, const char *name, u32 *out);

Có thêm bản _u8 / _u16 / _u64 cùng chữ ký, khác kiểu dữ liệu ra.

u32 val;
int ret = of_property_read_u32(np, "sample-rate-hz", &val);
if (ret)
    val = 1000; /* mặc định nếu DT không khai */

0 = thành công  ·  -EINVAL = không có property  ·  -EOVERFLOW = sai kích thước

int of_property_read_u32_array(const struct device_node *np, const char *name, u32 *out, size_t sz);

Đọc cả mảng cùng lúc. Nếu chỉ cần một phần tử theo vị trí, dùng of_property_read_u32_index(np, name, index, &out) thay vì đọc hết rồi tự index.

u32 ranges[3];
of_property_read_u32_array(np, "accel-ranges", ranges, 3);

property — chuỗi

int of_property_read_string(const struct device_node *np, const char *name, const char **out);

Con trỏ out trỏ thẳng vào vùng nhớ của device tree blob — không cần free, nhưng cũng không được sửa.

const char *mode;
of_property_read_string(np, "power-mode", &mode);

Property kiểu danh sách chuỗi thì dùng of_property_read_string_index().

property — boolean

bool of_property_read_bool(const struct device_node *np, const char *name);

Khác hẳn nhóm trên — trả thẳng bool, không qua tham số ra. Chỉ cần property tồn tại (không cần = <...>) là true, dùng cho property kiểu cờ.

bool wakeup = of_property_read_bool(np, "wakeup-source");

tài nguyên — gpio

struct gpio_desc *devm_gpiod_get(struct device *dev, const char *con_id, enum gpiod_flags flags);

con_id khớp với tiền tố property trong DT — "reset" ứng với reset-gpios.

#include <linux/gpio/consumer.h>

struct gpio_desc *rst = devm_gpiod_get(dev, "reset", GPIOD_OUT_LOW);

DT tương ứng: reset-gpios = <&gpio 17 GPIO_ACTIVE_LOW>;

Hàm cũ hay gặp trong code cũ: of_get_named_gpio(np, "reset-gpios", 0) từ <linux/of_gpio.h>, trả về số GPIO thô kiểu int. Đã lỗi thời — driver mới nên dùng gpiod ở trên.

tài nguyên — irq

int of_irq_get(struct device_node *np, int index);
#include <linux/of_irq.h>

int irq = of_irq_get(np, 0);
if (irq == -EPROBE_DEFER)
    return irq; /* interrupt controller chưa sẵn sàng */
Riêng driver I2C: nếu node có interrupts = <...>;, I2C core tự điền sẵn client->irq — thường chỉ cần dùng thẳng client->irq, không phải gọi of_irq_get() thủ công. Đây chính là con đường để chuyển driver MPU6050 từ đọc theo chu kỳ (polling) sang đọc khi có ngắt từ chân INT.

tài nguyên — clock & power

struct clk *devm_clk_get(struct device *dev, const char *id); struct regulator *devm_regulator_get(struct device *dev, const char *id);
#include <linux/clk.h>
#include <linux/regulator/consumer.h>

struct clk *clk = devm_clk_get(dev, "ext-clk");
struct regulator *vdd = devm_regulator_get(dev, "vdd");

DT tương ứng: clocks = <&osc 0>; clock-names = "ext-clk"; và vdd-supply = <&reg_3v3>;

phandle tham chiếu tới node khác

struct device_node *of_parse_phandle(const struct device_node *np, const char *name, int index);
struct device_node *tgt = of_parse_phandle(np, "some-ref", 0);

Property kiểu tham chiếu <&node> sang node khác trong cây. Nếu phandle có kèm cell tham số (kiểu interrupts-extended, pinctrl), dùng of_parse_phandle_with_args() thay cho bản đơn giản này.

match_data một driver, nhiều biến thể chip

Gắn một payload .data riêng cho từng compatible string, rồi lấy lại đúng payload đó trong probe() — tránh phải if/else theo compatible thủ công.

#include <linux/of_device.h>

static const struct of_device_id mpu_of_match[] = {
    { .compatible = "ledat,mpu6050", .data = &mpu6050_cfg },
    { .compatible = "ledat,mpu9250", .data = &mpu9250_cfg },
    { }
};

const struct mpu_cfg *cfg = of_device_get_match_data(&client->dev);

of_platform_* driver tầng bus / bridge

Họ này không đọc property — nó tạo device. Dùng khi driver của bạn đứng vai cha của các node con trong DT: driver bus/bridge (I2C mux, SPI bridge), MFD (1 chip vật lý expose nhiều chức năng con — PMIC có regulator + RTC + GPIO expander chẳng hạn), hoặc node kiểu simple-bus.

int of_platform_populate(struct device_node *root, const struct of_device_id *matches, const struct of_dev_auxdata *lookup, struct device *parent);

Duyệt các node con của root, node nào compatible khớp matches thì tạo platform_device và đăng ký vào driver core — driver của node con đó lại được match & probe() y hệt quy trình MPU6050, chỉ là lồng thêm một tầng.

of_platform_populate(dev->of_node, NULL, NULL, dev);

matches thường truyền of_default_bus_match_table, hoặc NULL nếu chỉ populate đúng node top-level.

int of_platform_default_populate(struct device_node *root, const struct of_dev_auxdata *lookup, struct device *parent);

Bản rút gọn của hàm trên, tự dùng sẵn of_default_bus_match_table.

struct platform_device *of_platform_device_create(struct device_node *np, const char *bus_id, struct device *parent);

Tạo trực tiếp 1 platform_device cho đúng 1 node cụ thể — không duyệt đệ quy như populate, dùng khi biết chính xác node nào cần tạo.

void of_platform_depopulate(struct device *parent);

Gỡ các device đã tạo bởi populate — gọi trong remove(), đối xứng với populate ở probe().

Riêng bus I2C/SPI: không cần gọi of_platform_populate() cho các thiết bị con nằm ngay trên chính bus đó — I2C/SPI core tự làm việc này ngầm khi bạn i2c_add_adapter(). of_platform_* chỉ cần khi node con là platform_device (MFD, simple-bus), không phải i2c_client/spi_device.

Hai công cụ duyệt cây còn lại, hay dùng chung với nhóm trên:

struct device_node *of_find_compatible_node(struct device_node *from, const char *type, const char *compat);

Tìm 1 node bất kỳ trong toàn cây theo compatible — không giới hạn con trực tiếp của node hiện tại.

for_each_child_of_node(parent, child) { ... }

Macro duyệt qua từng node con để đọc, không tạo device gì cả. Thoát vòng lặp sớm bằng break thì phải tự of_node_put(child) — quên là leak refcount.

áp dụng vào MPU6050

Thêm một property tùy chọn cho dải đo gyro, có fallback về mặc định 250 °/s nếu DT không khai — chỉ 2 dòng chèn vào probe().

mpu6050@68 {
    compatible = "ledat,mpu6050";
    reg = <0x68>;
    invensense,gyro-range = <500>;
};
u32 gyro_dps = 250; /* mặc định nếu DT không khai */
of_property_read_u32(client->dev.of_node, "invensense,gyro-range", &gyro_dps);

u8 gyro_fs;
switch (gyro_dps) {
case 500:  gyro_fs = MPU6050_GYRO_FS_500;  break;
case 1000: gyro_fs = MPU6050_GYRO_FS_1000; break;
case 2000: gyro_fs = MPU6050_GYRO_FS_2000; break;
default:   gyro_fs = MPU6050_GYRO_FS_250;  break;
}
i2c_smbus_write_byte_data(client, MPU6050_GYRO_CONFIG, gyro_fs);
Quay lại sơ đồ runtime ở lượt trước: hai dòng này chính là phần diễn ra bên trong ô "probe() chạy" — DT chỉ thay đổi đầu vào cấu hình, còn thứ tự khớp compatible → probe → sysfs vẫn giữ nguyên.
